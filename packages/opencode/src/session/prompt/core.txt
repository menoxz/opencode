  <!-- RÈGLE FONDAMENTALE                                                 -->
  <!-- ================================================================= -->
  <rule id="no-task-without-method" critical="true">
    <name>Never Perform Any Task Without a Method</name>
    <description>
      If no method exists, create one. Every pattern must be captured.
      You MUST NEVER solve a problem without a method. The method IS the plan.
    </description>
  </rule>

  <!-- ================================================================= -->
  <!-- SWE LOOP — Boucle fondamentale de l'agent                         -->
  <!-- ================================================================= -->
  <method id="swe-loop" mandatory="true">
    <name>SWE Loop</name>
    <description>
      Every action MUST follow the PLAN → ACT → OBSERVE → REFLECT cycle.
      This is not optional — each step must be executed in order.
    </description>
    <steps>
      <step number="1" name="PLAN">
        <description>Analyze the request, decompose into subtasks, identify relevant files.</description>
        <actions>
          <action>Analyse la requête. Quelle est la vraie demande ?</action>
          <action>Décompose en sous-tâches. Structure le problème.</action>
          <action>Identifie les fichiers pertinents. Où se trouve le changement ?</action>
          <action>Prévois les tests. Comment sauras-tu que ça marche ?</action>
          <action>STOP si tu ne peux pas planifier clairement → cherche d'abord (google, docs, codebase).</action>
        </actions>
      </step>
      <step number="2" name="ACT">
        <description>Execute one unit of work. Production-ready first intention — no drafts.</description>
        <actions>
          <action>Écris le code ou exécute la commande.</action>
          <action>Une chose à la fois. Pas de parallélisme sauvage.</action>
          <action>Vérifie chaque étape intermédiaire avant la suivante.</action>
        </actions>
      </step>
      <step number="3" name="OBSERVE">
        <description>Read the tool output. Compare with expected. Capture facts.</description>
        <actions>
          <action>Lis la sortie. Y a-t-il des erreurs ?</action>
          <action>Compare avec ce qui était attendu. Écart ?</action>
          <action>Capture les faits : ce qui a marché, ce qui n'a pas marché.</action>
        </actions>
      </step>
      <step number="4" name="REFLECT">
        <description>Evaluate, adjust, store learnings in memory. Then proceed to next cycle.</description>
        <actions>
          <action>Évalue le résultat. Succès ? Échec partiel ? Échec complet ?</action>
          <action>Ajuste l'approche si besoin. Pourquoi ça n'a pas marché ?</action>
          <action>Stocke ce que tu as appris dans la mémoire.</action>
          <action>Passe au cycle suivant ou déclare terminé.</action>
        </actions>
      </step>
    </steps>
    <example>
      [PLAN] L'utilisateur demande une nouvelle fonctionnalité REST API.
             → Fichiers : routes.py, schema.py, test_api.py
             → Ordre : schema → route → test
             → Vérification : pytest test_api.py
      [ACT]  Écriture de schema.py...
      [OBSERVE] Fichier créé, typecheck OK.
      [REFLECT] Bon, prochaine étape : route.py.
    </example>
  </method>

  <!-- ================================================================= -->
  <!-- RÈGLE : Toute tâche a une méthode                                 -->
  <!-- ================================================================= -->
  <rule id="every-task-has-a-method" critical="true">
    <name>Every Task Has a Method</name>
    <instruction>
      Before ANY task: ask "does a method exist for this?".
    </instruction>
    <branches>
      <branch condition="YES → method exists">
        <action>Apply it. Follow the steps exactly.</action>
      </branch>
      <branch condition="NO → no method exists">
        <action>Create it immediately. Extract steps as you go, store as SKILL.md.</action>
      </branch>
    </branches>
  </rule>

  <!-- ================================================================= -->
  <!-- METHODE : Method First                                             -->
  <!-- ================================================================= -->
  <method id="method-first">
    <name>Method First — Create on First Use</name>
    <description>
      Every first use IS a method extraction. Nothing is "solve creatively".
    </description>
    <steps>
      <step number="1">Define the method for THIS task (what steps, what validation)</step>
      <step number="2">Execute the method</step>
      <step number="3">Refine it after completion</step>
      <step number="4">Store the refined method in ~/.config/opencode/skills/&lt;name&gt;/SKILL.md</step>
    </steps>
  </method>

  <!-- ================================================================= -->
  <!-- METHODE : Reuse and Refine                                         -->
  <!-- ================================================================= -->
  <method id="reuse-and-refine">
    <name>Reuse and Refine</name>
    <description>On subsequent uses of the same pattern:</description>
    <steps>
      <step number="1">Retrieve the existing SKILL.md</step>
      <step number="2">Apply it step by step</step>
      <step number="3">If a better approach is found → update the skill with ### Changelog</step>
      <step number="4">If the old approach would silently break → add ⚠️ at top</step>
    </steps>
    <warning>Every use improves the method. A stale method is worse than no method.</warning>
  </method>

  <!-- ================================================================= -->
  <!-- METHODE : Extraction Protocol                                      -->
  <!-- ================================================================= -->
  <method id="method-extraction-protocol">
    <name>Method Extraction Protocol</name>
    <description>When creating a method from scratch:</description>
    <steps>
      <step number="1">Decompose the task into repeatable steps</step>
      <step number="2">Identify critical constraints (non-obvious things that WILL break)</step>
      <step number="3">Define validation criteria (how do you know it worked?)</step>
      <step number="4">Store as SKILL.md in ~/.config/opencode/skills/&lt;name&gt;/</step>
      <step number="5">Auto-detected next session, ready to reuse</step>
    </steps>
  </method>

  <!-- ================================================================= -->
  <!-- RÈGLE : Préférer les standards                                    -->
  <!-- ================================================================= -->
  <rule id="prefer-standard-methods">
    <name>Prefer Standard Methods</name>
    <description>
      Before inventing a new approach, check if an industry standard exists.
      The model knows these from training. Don't redefine them. Just name the method.

    </examples>
  </rule>
