<task-instruction type="architecture">
  <name>Architecture Design</name>
  <description>Design system architecture with proper trade-off analysis.</description>

  <phase name="research" required="true">
    <name>FIRST STEP — Research the state of the art</name>
    <instruction>Before proposing an architecture:</instruction>
    <actions>
      <action>Search for reference architectures for this type of system</action>
      <action>Find approach comparisons (microservices vs monolith, etc.)</action>
      <action>Check limitations and capabilities of considered technologies</action>
      <action>Review existing RFCs and ADRs</action>
    </actions>
  </phase>

  <workflow>
    <step number="1" icon="🔍" name="Research">What similar architectures exist?</step>
    <step number="2" icon="📋" name="Requirements">Understand functional and technical constraints</step>
    <step number="3" icon="🚫" name="Constraints">Identify limits (budget, time, skills, infrastructure)</step>
    <step number="4" icon="🏗️" name="Design">Propose 2-3 approaches with their trade-offs</step>
    <step number="5" icon="📝" name="Documentation">ADR, diagrams, key decisions</step>
    <step number="6" icon="🛣️" name="Migration plan">How to go from current state to target</step>
  </workflow>

  <principles>
    <principle>Compare at least 2 approaches before recommending</principle>
    <principle>Consider future evolution, not just the immediate need</principle>
    <principle>Document decisions and their justifications (ADR)</principle>
    <principle>Favor simplicity unless complexity is justified</principle>
  </principles>
</task-instruction>
