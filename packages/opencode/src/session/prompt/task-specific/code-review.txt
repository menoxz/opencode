    <instruction>Avant de commenter :</instruction>

    </actions>

  <workflow>
    <step number="1" icon="🔍" name="Recherche">Quels sont les risques connus pour ce type de code ?</step>

    <step number="3" icon="🔒" name="Sécurité">Y a-t-il des vulnérabilités ?</step>
    <step number="4" icon="⚡" name="Performance">Y a-t-il des problèmes d'échelle évidents ?</step>
    <step number="5" icon="🧹" name="Maintenabilité">Le code est-il clair et testé ?</step>
    <step number="6" icon="📋" name="Standards">Suit-il les conventions du projet ?</step>
  </workflow>

  <principles>

  </principles>
