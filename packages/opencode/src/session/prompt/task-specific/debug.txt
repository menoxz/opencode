<task-instruction type="debug">
  <name>Systematic Debugging</name>
  <description>Systematically diagnose and resolve errors.</description>

  <phase name="research" required="true">
    <name>FIRST STEP — Search for the error online</name>
    <instruction>Before experimenting, search the web for:</instruction>
    <actions>
      <action>The exact error message (full stack trace)</action>
      <action>GitHub issues with the same stack trace</action>
      <action>Relevant API / framework documentation</action>
      <action>Known solutions on Stack Overflow, forums, etc.</action>
    </actions>
  </phase>

  <protocol name="systematic-debugging">
    <step number="1" icon="🔍" name="Search error">Search the error online — don't guess, Google is your friend</step>
    <step number="2" icon="📖" name="Understand">Read the full stack trace and error message</step>
    <step number="3" icon="🔬" name="Reproduce">Create a minimal reproduction case</step>
    <step number="4" icon="🎯" name="Isolate">Binary search / divide and conquer</step>
    <step number="5" icon="💡" name="Hypothesis">Expected vs actual behavior</step>
    <step number="6" icon="🧪" name="Test hypothesis">One experiment at a time</step>
    <step number="7" icon="🔧" name="Fix">Apply the minimal correction</step>
    <step number="8" icon="✅" name="Verify">Confirm the fix + no regressions</step>
  </protocol>

  <common-checks>
    <check type="type-errors">Check interfaces, generics, null/undefined</check>
    <check type="runtime-errors">Check async flows, error boundaries, edge cases</check>
    <check type="logic-errors">Trace through the algorithm step by step</check>
    <check type="integration-errors">Check API contracts, data formats</check>
    <check type="performance-issues">Profile before optimizing</check>
  </common-checks>
</task-instruction>
