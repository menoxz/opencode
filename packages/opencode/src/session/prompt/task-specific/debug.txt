<task-instruction type="debug">
  <name>Systematic Debugging</name>
  <description>Systematically diagnose and resolve errors.</description>

  <phase name="research" required="true">
    <name>FIRST STEP — Cherche l'erreur en ligne</name>
    <instruction>Avant toute expérimentation, cherche sur le web :</instruction>
    <actions>

      <action>Des issues GitHub avec la même stack trace</action>
      <action>La documentation de l'API / framework concerné</action>

    </actions>
  </phase>

  <protocol name="systematic-debugging">
    <step number="1" icon="🔍" name="Recherche erreur">Cherche l'erreur en ligne — ne devine pas, Google est ton ami</step>
    <step number="2" icon="📖" name="Compréhension">Lis le stack trace et le message complet</step>
    <step number="3" icon="🔬" name="Reproduction">Crée un cas de reproduction minimal</step>
    <step number="4" icon="🎯" name="Isolement">Binary search / divide and conquer</step>
    <step number="5" icon="💡" name="Hypothèse">Comportement attendu vs réel</step>
    <step number="6" icon="🧪" name="Test hypothèse">Une expérience à la fois</step>
    <step number="7" icon="🔧" name="Correction">Applique la correction minimale</step>
    <step number="8" icon="✅" name="Vérification">Confirme le fix + pas de régressions</step>
  </protocol>

  <common-checks>
    <check type="type-errors">Check interfaces, generics, null/undefined</check>
    <check type="runtime-errors">Check async flows, error boundaries, edge cases</check>
    <check type="logic-errors">Trace through the algorithm step by step</check>
    <check type="integration-errors">Check API contracts, data formats</check>
    <check type="performance-issues">Profile before optimizing</check>
  </common-checks>
</task-instruction>
