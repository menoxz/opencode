<task-instruction type="bug-fix">
  <name>Bug Fix</name>
  <description>Focus on finding and fixing the root cause.</description>

  <phase name="research" required="true">
    <name>FIRST STEP — Cherche le bug en ligne</name>
    <instruction>Avant de toucher au code, cherche sur le web :</instruction>
    <actions>
      <action>L'erreur exacte (stack trace, message)</action>
      <action>Des issues GitHub similaires</action>
      <action>La documentation de l'API concernée</action>
      <action>Des solutions connues pour ce pattern d'erreur</action>
    </actions>
  </phase>

  <workflow>
    <step number="1" icon="🔍" name="Recherche symptôme">Cherche le symptôme en ligne — quelqu'un a déjà eu ce bug</step>
    <step number="2" icon="🔬" name="Reproduction">Comprends l'échec exact</step>
    <step number="3" icon="🎯" name="Isolement">Isole la root cause — utilise une reproduction minimale</step>
    <step number="4" icon="🔧" name="Correction">Applique le fix minimal — change UNIQUEMENT ce qui est nécessaire</step>
    <step number="5" icon="✅" name="Vérification">Confirme que le fix marche ET que le comportement existant est préservé</step>
  </workflow>

  <principles>
    <principle>Préfère le fix le plus simple qui adresse la root cause</principle>
    <principle>Ajoute des tests de régression</principle>
    <principle>Si le bug est dans une dépendance, considère un workaround d'abord</principle>
    <principle>Documente la root cause et le rationale du fix</principle>
  </principles>
</task-instruction>
