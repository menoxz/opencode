<task-instruction type="bug-fix">
  <name>Bug Fix</name>
  <description>Focus on finding and fixing the root cause.</description>

  <phase name="research" required="true">
    <name>FIRST STEP — Search for the bug online</name>
    <instruction>Before touching any code, search the web for:</instruction>
    <actions>
      <action>The exact error (stack trace, message)</action>
      <action>Similar GitHub issues</action>
      <action>Relevant API documentation</action>
      <action>Known solutions for this error pattern</action>
    </actions>
  </phase>

  <workflow>
    <step number="1" icon="🔍" name="Symptom research">Search online for the symptom — someone has likely hit this bug before</step>
    <step number="2" icon="🔬" name="Reproduction">Understand the exact failure</step>
    <step number="3" icon="🎯" name="Isolation">Isolate the root cause using minimal reproduction</step>
    <step number="4" icon="🔧" name="Fix">Apply the minimal fix — change ONLY what is necessary</step>
    <step number="5" icon="✅" name="Verification">Confirm the fix works AND existing behavior is preserved</step>
  </workflow>

  <principles>
    <principle>Prefer the simplest fix that addresses the root cause</principle>
    <principle>Add regression tests</principle>
    <principle>If the bug is in a dependency, consider a workaround first</principle>
    <principle>Document the root cause and fix rationale</principle>
  </principles>
</task-instruction>
